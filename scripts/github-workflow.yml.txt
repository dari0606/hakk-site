# Готовый файл для GitHub Actions.
# Создайте его в репозитории по пути .github/workflows/youtube-sync.yml
# (Add file -> Create new file на github.com) и вставьте всё, что ниже.

# Раз в сутки забирает с YouTube-канала новые видео и перепроверяет,
# какие уроки открыты. Если что-то изменилось — коммитит, и Vercel
# сам пересобирает сайт.
name: Синхронизация с YouTube

on:
  schedule:
    - cron: '0 3 * * *'   # 03:00 UTC = 08:00 по Алматы
  workflow_dispatch:       # и кнопкой вручную на вкладке Actions

jobs:
  sync:
    runs-on: ubuntu-latest
    permissions:
      contents: write
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 22

      - name: Новые видео с канала
        run: npm run sync

      - name: Проверить, какие уроки открыты
        run: npm run check

      - name: Записать изменения
        run: |
          if [ -n "$(git status --porcelain data/lessons.json)" ]; then
            git config user.name "Hakk Academy bot"
            git config user.email "orumbayevaaaa@gmail.com"
            git add data/lessons.json
            git commit -m "YouTube: обновление каталога уроков"
            git push
            echo "Каталог обновлён — Vercel пересоберёт сайт."
          else
            echo "Изменений нет."
          fi
